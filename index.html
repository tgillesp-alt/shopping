<script>

const SUPABASE_URL = 'https://izdclsyuplghwznsjnom.supabase.co';

const SUPABASE_KEY = 'eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJpc3MiOiJzdXBhYmFzZSIsInJlZiI6Iml6ZGNsc3l1cGxnaHd6bnNqbm9tIiwicm9sZSI6ImFub24iLCJpYXQiOjE3ODE1MjY5MjksImV4cCI6MjA5NzEwMjkyOX0.GkjxcToCHZrLAm1pVGsBkcLapIOj7uSYd6IcDC0Lyzk';

const PASSORD = 'fager2026';


let bruker = '';

let varer = [];

let aktivTab = 'aktive';


function loggInn() {

  const navn = document.getElementById('login-navn').value.trim();

  const passord = document.getElementById('login-passord').value;

  if (passord !== PASSORD) {

    document.getElementById('login-error').style.display = 'block';

    return;

  }

  if (!navn) { alert('Skriv inn navn'); return; }

  bruker = navn;

  document.getElementById('login-screen').style.display = 'none';

  document.getElementById('app').style.display = 'block';

  document.getElementById('header-bruker').textContent = bruker;

  hent();

  setInterval(hent, 15000);

}


async function api(method, path, body) {

  const opts = {

    method,

    headers: {

      'apikey': SUPABASE_KEY,

      'Authorization': 'Bearer ' + SUPABASE_KEY,

      'Content-Type': 'application/json',

      'Prefer': method === 'POST' ? 'return=representation' : ''

    }

  };

  if (body) opts.body = JSON.stringify(body);

  const res = await fetch(SUPABASE_URL + '/rest/v1/' + path, opts);

  if (res.status === 204) return [];

  return res.json();

}


async function hent() {

  varer = await api('GET', 'varer?order=created_at.desc&select=*');

  render();

}


function byttTab(tab, el) {

  aktivTab = tab;

  document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));

  el.classList.add('active');

  render();

}


function render() {

  const filterLev = document.getElementById('filter-lev').value;

  const filterKat = document.getElementById('filter-kat').value;

  const sok = document.getElementById('filter-sok').value.toLowerCase();


  const aktive = varer.filter(v => !v.bestilt && !v.mottatt);

  const bestilt = varer.filter(v => v.bestilt && !v.mottatt);

  const mottatt = varer.filter(v => v.mottatt);


  document.getElementById('count-aktive').textContent = aktive.length;

  document.getElementById('count-bestilt').textContent = bestilt.length;

  document.getElementById('count-mottatt').textContent = mottatt.length;


  let vis = aktivTab === 'aktive' ? aktive : aktivTab === 'bestilt' ? bestilt : mottatt;


  if (filterLev) vis = vis.filter(v => v.leverandor === filterLev);

  if (filterKat) vis = vis.filter(v => v.kategori === filterKat);

  if (sok) vis = vis.filter(v => v.navn.toLowerCase().includes(sok));


  const grouped = {};

  vis.forEach(v => {

    const key = v.leverandor || v.kategori || 'Ukategorisert';

    if (!grouped[key]) grouped[key] = [];

    grouped[key].push(v);

  });


  const container = document.getElementById('liste');

  if (vis.length === 0) {

    container.innerHTML = '<div class="empty">Ingen varer</div>';

    return;

  }


  container.innerHTML = Object.entries(grouped).sort((a,b) => a[0].localeCompare(b[0])).map(([group, items]) => `

    <div class="section">

      <div class="section-header">${group}</div>

      ${items.map(v => vareHTML(v)).join('')}

    </div>

  `).join('');

}


function vareHTML(v) {

  const cls = v.mottatt ? 'mottatt' : v.bestilt ? 'bestilt' : '';

  const meta = [v.kategori, v.mengde ? v.mengde + (v.enhet ? ' ' + v.enhet : '') : '', v.lagt_inn_av ? 'av ' + v.lagt_inn_av : ''].filter(Boolean);

  return `

    <div class="vare ${cls}" id="vare-${v.id}">

      <div class="vare-check">

        <label><input type="checkbox" ${v.bestilt || v.mottatt ? 'checked' : ''} onchange="toggleBestilt('${v.id}', this.checked)"> Bestilt</label>

        <label><input type="checkbox" ${v.mottatt ? 'checked' : ''} onchange="toggleMottatt('${v.id}', this.checked)"> Mottatt</label>

      </div>

      <div class="vare-info">

        <div class="vare-navn">${v.navn}</div>

        <div class="vare-meta">${meta.map(m => `<span>${m}</span>`).join('')}</div>

      </div>

      <button class="vare-del" onclick="slett('${v.id}')" title="Slett">✕</button>

    </div>

  `;

}


async function leggTil() {

  const navn = document.getElementById('ny-navn').value.trim();

  const kat = document.getElementById('ny-kat').value;

  if (!navn || !kat) { alert('Fyll inn varenavn og kategori'); return; }

  await api('POST', 'varer', {

    navn,

    kategori: kat,

    leverandor: document.getElementById('ny-lev').value || null,

    mengde: document.getElementById('ny-mengde').value || null,

    enhet: document.getElementById('ny-enhet').value || null,

    lagt_inn_av: bruker,

    bestilt: false,

    mottatt: false

  });

  document.getElementById('ny-navn').value = '';

  document.getElementById('ny-mengde').value = '';

  document.getElementById('ny-enhet').value = '';

  hent();

}


async function toggleBestilt(id, checked) {

  await api('PATCH', `varer?id=eq.${id}`, { bestilt: checked, mottatt: checked ? false : undefined });

  hent();

}


async function toggleMottatt(id, checked) {

  await api('PATCH', `varer?id=eq.${id}`, { mottatt: checked, bestilt: checked ? true : undefined });

  hent();

}


async function slett(id) {

  if (!confirm('Slett denne varen?')) return;

  await api('DELETE', `varer?id=eq.${id}`);

  hent();

}


document.getElementById('login-passord').addEventListener('keydown', e => { if (e.key === 'Enter') loggInn(); });
